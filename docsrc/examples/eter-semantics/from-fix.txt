// Imm to Imm

    let imm x: i32 = 1;
    let imm y: i32 = x;
    f(x);
}

// ==========

    let imm x: i32 = 1;

struct Point {
    imm x: i32,
    imm y: i32,
}

// ==========
// Imm to Imm
// ==========
fn f(imm p: Point) {} // Drop(p)
fn main() {
    let imm p = Point { x: 1, y: 2 };
    let imm q = p;
    f(p);
} // Drop(q), Drop(p)

// ==========
// Imm to Mut
// ==========
fn f(mut p: Point) {} // Drop(p)
fn main() {
    let imm p = Point { x: 1, y: 2 };
    // let mut q = p; // ERROR: Changing the regime from imm to mut is not allowed. Need to copy.
    // f(p); // ERROR: Changing the regime from imm to mut is not allowed. Need to copy.
}

// ===========
// Imm to Proj
// ===========
fn f(proj p: Point) {} // (proj parameter, no Drop)
fn main() {
    let imm p = Point { x: 1, y: 2 };
    // let proj q = &p; // ERROR: Changing the regime from imm to proj is not allowed. Need to copy.
    // f(&p); // ERROR: Changing the regime from imm to proj is not allowed. Need to copy.
}

    let imm p = Point { x: 1, y: 2 };

    let imm q = f2(p1, p2);

    imm x: i32,
    imm y: i32,

    let imm y = p.y;
    Point { x, y }

    let imm p1 = Point { x: 1, y: 2 };
    let imm p2 = f(p1);
} // Drop(p2), Drop(p1)
